# A2A Task Lifecycle Implementation Plan

> **For Hermes:** Use the a2a-hermes-opencode skill to dispatch this plan to OpenCode task-by-task.

**Goal:** Add a Google A2A-spec Task state machine, persistent session storage, Agent Registry, and Task Orchestrator to `qa_framework/a2a_adapter.py` so long-running QA tasks survive interruptions and dispatch to matched agents.

**Architecture:** 5 new dataclasses added to the existing ~400-line adapter. TaskRecord mirrors A2A Task spec (id, contextId, status, history, artifacts). TaskStateMachine validates explicit state transitions. TaskSession checkpoints running state to disk every 30s with auto-recovery. AgentRegistry parses Agent Cards and matches tasks to agents. TaskOrchestrator supports fan-out/fan-in and pipeline patterns.

**Tech Stack:** Python 3.14 stdlib only — dataclasses, threading, json, hashlib, pathlib, datetime, typing.

```python
from dataclasses import dataclass, field
from datetime import datetime, timezone
from typing import Optional
import json
import hashlib

@dataclass
class TaskRecord:
    """Mirrors Google A2A Task spec."""
    task_id: str
    task_type: str = "generic"
    context_id: Optional[str] = None
    status: str = "submitted"
    history: list = field(default_factory=list)
    artifacts: list = field(default_factory=list)
    created_at: str = field(default_factory=lambda: datetime.now(timezone.utc).isoformat())
    updated_at: str = field(default_factory=lambda: datetime.now(timezone.utc).isoformat())
    error: Optional[str] = None
    
    def transition_to(self, new_status: str) -> None:
        """Transition to new status, validating with state machine."""
        sm = TaskStateMachine()
        if not sm.can_transition(self.status, new_status):
            raise ValueError(f"Invalid transition: {self.status} → {new_status}")
        self.status = new_status
        self.updated_at = datetime.now(timezone.utc).isoformat()
    
    def add_artifact(self, artifact_id: str, content: str, revision: int = 0) -> dict:
        """Add an artifact to this task."""
        artifact = {
            "artifact_id": artifact_id,
            "content": content,
            "revision": revision,
            "created_at": datetime.now(timezone.utc).isoformat(),
        }
        self.artifacts.append(artifact)
        self.updated_at = datetime.now(timezone.utc).isoformat()
        return artifact
    
    def to_dict(self) -> dict:
        """Serialize to dict."""
        return {
            "task_id": self.task_id,
            "task_type": self.task_type,
            "context_id": self.context_id,
            "status": self.status,
            "history": self.history,
            "artifacts": self.artifacts,
            "created_at": self.created_at,
            "updated_at": self.updated_at,
            "error": self.error,
        }
    
    def to_json(self) -> str:
        return json.dumps(self.to_dict(), indent=2)


class TaskStateMachine:
    """A2A Task lifecycle state machine."""
    
    def __init__(self):
        self.states = {"submitted", "working", "input-required", "completed", "failed", "canceled"}
        # Allowed transitions
        self._transitions = {
            "submitted": {"working", "canceled"},
            "working": {"completed", "failed", "input-required"},
            "input-required": {"working", "canceled"},
            "completed": set(),  # terminal
            "failed": set(),     # terminal
            "canceled": set(),   # terminal
        }
    
    def can_transition(self, from_state: str, to_state: str) -> bool:
        if from_state not in self.states or to_state not in self.states:
            return False
        return to_state in self._transitions.get(from_state, set())
```

```python
import os

@dataclass
class TaskSession:
    """Checkpoints running task state to disk for crash recovery."""
    session_id: str
    checkpoint_dir: str
    _lock: threading.Lock = field(default_factory=threading.Lock, repr=False)
    
    def __post_init__(self):
        os.makedirs(self.checkpoint_dir, exist_ok=True)
    
    def _task_path(self, task_id: str) -> str:
        return os.path.join(self.checkpoint_dir, f"{self.session_id}_{task_id}.json")
    
    def save_checkpoint(self, task: TaskRecord) -> None:
        with self._lock:
            path = self._task_path(task.task_id)
            with open(path, "w") as f:
                json.dump(task.to_dict(), f, indent=2)
    
    def load_checkpoint(self, task_id: str) -> TaskRecord:
        path = self._task_path(task_id)
        with open(path) as f:
            data = json.load(f)
        return TaskRecord(**data)
    
    def list_recoverable_tasks(self) -> list[TaskRecord]:
        """List tasks that were working and can be recovered."""
        result = []
        with self._lock:
            for fname in os.listdir(self.checkpoint_dir):
                if fname.startswith(f"{self.session_id}_") and fname.endswith(".json"):
                    path = os.path.join(self.checkpoint_dir, fname)
                    with open(path) as f:
                        data = json.load(f)
                    t = TaskRecord(**data)
                    if t.status in ("working", "input-required"):
                        result.append(t)
        return result
    
    def delete_checkpoint(self, task_id: str) -> None:
        path = self._task_path(task_id)
        if os.path.exists(path):
            os.remove(path)
```

```python
@dataclass
class AgentRegistry:
    """Agent Card registry — matches tasks to agents by capability."""
    _agents: dict[str, dict] = field(default_factory=dict)
    _lock: threading.Lock = field(default_factory=threading.Lock, repr=False)
    
    def register(self, agent_id: str, card: dict) -> None:
        with self._lock:
            self._agents[agent_id] = card
    
    def unregister(self, agent_id: str) -> None:
        with self._lock:
            self._agents.pop(agent_id, None)
    
    def can_handle(self, agent_id: str, task_type: str) -> bool:
        card = self._agents.get(agent_id, {})
        skills = card.get("skills", [])
        capabilities = card.get("capabilities", [])
        return task_type in skills or task_type in capabilities
    
    def best_agent(self, task_type: str) -> Optional[str]:
        """Find best agent for task type (most specialized match first)."""
        candidates = []
        for agent_id, card in self._agents.items():
            skills = card.get("skills", [])
            capabilities = card.get("capabilities", [])
            if task_type in skills:
                candidates.append((agent_id, len(skills)))  # fewer skills = more specialized
            elif task_type in capabilities:
                candidates.append((agent_id, len(capabilities) + 100))
        if not candidates:
            return None
        # Pick agent with fewest total skills (most specialized)
        candidates.sort(key=lambda x: x[1])
        return candidates[0][0]
    
    def list_agents(self) -> list[str]:
        return list(self._agents.keys())
```

```python
@dataclass
class TaskOrchestrator:
    """Orchestrate multi-agent task execution patterns."""
    
    def fan_out(self, tasks: list[TaskRecord], dispatch_fn) -> list:
        """Dispatch tasks to parallel workers, collect results sequentially."""
        results = []
        for t in tasks:
            try:
                result = dispatch_fn(t)
                results.append(result)
            except Exception as e:
                t.transition_to("failed")
                t.error = str(e)
                results.append((t.task_id, f"error: {e}"))
        return results
    
    def pipeline(self, task: TaskRecord, stages: list[tuple]) -> TaskRecord:
        """Execute sequential stages on a task. Each stage is (fn, success_status)."""
        for fn, success_status in stages:
            fn(task)
            if success_status:
                task.transition_to(success_status)
        return task
```
